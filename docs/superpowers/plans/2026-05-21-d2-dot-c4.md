# D2 + DOT C4 Coverage Implementation Plan (Wave G)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add c4 importer + exporter to the `DiagramKitD2` and `DiagramKitGraphviz` slices, taking D2 to 9/28 and DOT to 8/28 in each direction, all mediated by the canonical `C4Diagram` payload and comment-encoded recovery markers.

**Architecture:** Each format gets a probe (`{D2,DOT}C4Probe`), a mapper (`{D2,DOT}C4Mapper`), and an exporter (`{D2,DOT}C4Export`) that talk only to the existing `C4Diagram` payload (`Sources/DiagramKitModel/src_c4_types.swift`). Eleven new shared marker case names extend each format's `RecoveryMarker.Kind` enum. Cross-format pairs (mermaid/plantuml/structurizr ↔ d2/dot, plus d2 ↔ dot) round-trip through the canonical payload — no format-specific bridges.

**Tech Stack:** Swift 6, swift-testing, swift-snapshot-testing, `DiagramKitCommon.RecoveryMarkerScanner`, `DiagramKitTestSupport.RoundTripHarness`.

**Spec:** [`docs/superpowers/specs/2026-05-21-d2-dot-c4-design.md`](../specs/2026-05-21-d2-dot-c4-design.md) (commit `fedd423`).

import Foundation
import DiagramKitCommon
import DiagramKitModel

public enum D2C4Export {

    public static func export(_ diagram: C4Diagram) -> String {
        var lines: [String] = []
        lines.append(D2RecoveryMarker.emitFamily("c4"))
        lines.append(D2RecoveryMarker.emitC4DiagramKind(diagram.kind.rawValue))
        if let title = diagram.title, !title.isEmpty {
            lines.append("title: \"\(escape(title))\"")
        }
        lines.append("")

        for shape in diagram.shapes where shape.parentBoundary.isEmpty {
            emitShape(shape, indent: "", into: &lines)
        }

        return lines.joined(separator: "\n") + "\n"
    }

    static func emitShape(_ shape: C4Shape, indent: String, into lines: inout [String]) {
        let (nativeShape, needsKindMarker) = nativeD2Shape(for: shape.typeC4Shape)
        lines.append("\(indent)\(shape.alias): \"\(escape(shape.label))\" {")
        lines.append("\(indent)    shape: \(nativeShape)")
        lines.append("\(indent)}")
        if needsKindMarker {
            lines.append(D2RecoveryMarker.emitC4ShapeKind(
                targetID: shape.alias,
                rawValue: shape.typeC4Shape.rawValue
            ))
        }
        if shape.typeC4Shape.isExternal {
            lines.append(D2RecoveryMarker.emitC4External(targetID: shape.alias))
        }
    }

    /// Returns (native D2 shape name, whether c4-shape-kind marker is required to recover the original C4ShapeType).
    static func nativeD2Shape(for type: C4ShapeType) -> (String, Bool) {
        switch type {
        case .person, .external_person:                                  return ("person",    false)
        case .system, .system_ext:                                       return ("rectangle", false)
        case .system_db, .system_db_ext:                                 return ("cylinder",  false)
        case .system_queue, .system_queue_ext:                           return ("queue",     false)
        case .container, .container_ext:                                 return ("rectangle", true)
        case .container_db, .container_db_ext:                           return ("cylinder",  true)
        case .container_queue, .container_queue_ext:                     return ("queue",     true)
        case .component, .component_ext:                                 return ("hexagon",   false)
        case .component_db, .component_db_ext:                           return ("cylinder",  true)
        case .component_queue, .component_queue_ext:                     return ("queue",    true)
        }
    }

    private static func escape(_ s: String) -> String {
        s.replacingOccurrences(of: "\"", with: "\\\"")
    }
}

private extension C4ShapeType {
    var isExternal: Bool {
        switch self {
        case .external_person, .system_ext, .system_db_ext, .system_queue_ext,
             .container_ext, .container_db_ext, .container_queue_ext,
             .component_ext, .component_db_ext, .component_queue_ext:
            return true
        default:
            return false
        }
    }
}

        var lines: [String] = []
        lines.append(D2RecoveryMarker.emitFamily("c4"))
        lines.append(D2RecoveryMarker.emitC4DiagramKind(diagram.kind.rawValue))
        if let title = diagram.title, !title.isEmpty {
            lines.append("title: \"\(escape(title))\"")
        }
        lines.append("")

        let authoredBoundaries = diagram.boundaries.filter { $0.origin != .viewScopeSynthesized }
        let rootBoundaries = authoredBoundaries.filter { $0.parentBoundary.isEmpty }
        let childrenByParent = Dictionary(grouping: authoredBoundaries.filter { !$0.parentBoundary.isEmpty }, by: \.parentBoundary)
        let shapesByBoundary = Dictionary(grouping: diagram.shapes, by: \.parentBoundary)

        for boundary in rootBoundaries {
            emitBoundary(boundary, childrenByParent: childrenByParent, shapesByBoundary: shapesByBoundary, indent: "", into: &lines)
        }
        for shape in shapesByBoundary[""] ?? [] {
            emitShape(shape, indent: "", into: &lines)
        }

        return lines.joined(separator: "\n") + "\n"
    }

    private static func emitBoundary(
        _ boundary: C4Boundary,
        childrenByParent: [String: [C4Boundary]],
        shapesByBoundary: [String: [C4Shape]],
        indent: String,
        into lines: inout [String]
    ) {

        for child in childrenByParent[boundary.alias] ?? [] {
            emitBoundary(child, childrenByParent: childrenByParent, shapesByBoundary: shapesByBoundary, indent: indent + "    ", into: &lines)

        for shape in shapesByBoundary[boundary.alias] ?? [] {

import DiagramKitCommon

enum D2C4Probe {
    /// Returns true when the source declares `# diagramkit:family=c4`.
    /// Structural fallback: ≥1 `c4-shape-kind` or `c4-diagram-kind` marker present.
    static func detectsC4(_ source: String) -> Bool {
        let scan = D2RecoveryMarker.scanner.scan(source: source)
        if scan.markers.contains(where: {
            if case .family(let name) = $0.kind { return name == "c4" } else { return false }
        }) { return true }
        return scan.markers.contains {
            if case .c4DiagramKind = $0.kind { return true }
            if case .c4ShapeKind   = $0.kind { return true }
            return false
        }
    }
}

import DiagramKitCommon

enum D2C4Mapper {

    static func map(source: String) -> DiagramImportResult {
        var diagnostics: [DiagramDiagnostic] = []
        let scan = D2RecoveryMarker.scanner.scan(source: source)
        let markers = scan.markers

        let diagramKind = decodeDiagramKind(from: markers, diagnostics: &diagnostics)

        // Walk the D2 source line-by-line: open `<id>: "<label>" {` and close `}`
        // tracks boundary nesting. Lines that contain `shape: <name>` set the
        // current declaration's native shape. Lines like `a -> b: "label"` /
        // `a <-> b` become relationships.
        var ctx = ParseContext()
        for (lineIndex, rawLine) in source.split(omittingEmptySubsequences: false, whereSeparator: { $0.isNewline }).enumerated() {
            let line = rawLine.trimmingCharacters(in: .whitespaces)
            if line.isEmpty || line.hasPrefix("#") { continue }
            // … (declaration open / shape: / close / arrow handling)
        }

        ctx.applyMarkers(markers)

        let diagram = C4Diagram(
            kind: diagramKind,
            title: ctx.title,
            shapes: ctx.shapes,
            boundaries: ctx.boundaries,
            relationships: ctx.relationships,
            config: C4DiagramConfig()
        )
        return DiagramImportResult(
            document: DiagramDocument(payload: .c4(diagram)),
            diagnostics: diagnostics
        )
    }

    private struct ParseContext {
        var title: String?
        var boundaryStack: [String] = []
        var shapes: [C4Shape] = []
        var boundaries: [C4Boundary] = []
        var relationships: [C4Relationship] = []
        var declarationLineByTargetID: [String: Int] = [:]
        var edgeIndex: Int = 0
        mutating func applyMarkers(_ markers: [RecoveryMarker<D2RecoveryMarker.Kind>]) {
            // For each marker: look up the most recent declaration whose line
            // number is < marker.lineNumber, then mutate that shape/boundary/rel
            // to set the corresponding field.
        }
    }

    private static func decodeDiagramKind(
        from markers: [RecoveryMarker<D2RecoveryMarker.Kind>],
        diagnostics: inout [DiagramDiagnostic]
    ) -> C4DiagramKind {
        for m in markers {
            if case .c4DiagramKind(let raw) = m.kind, let kind = C4DiagramKind(rawValue: raw) {
                return kind
            }
        }
        diagnostics.append(.informational(.shapeDowngrade, "C4 diagram kind defaulted to .context (no c4-diagram-kind marker)"))
        return .context
    }
}
